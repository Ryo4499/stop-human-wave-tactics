server {
        listen 80;

        server_name ${DOMAIN};
        root /usr/share/nginx/html;

        location ~ ^/.well-known {
                root /usr/share/nginx/html;
        }

        location / {
                root /usr/share/nginx/html;
        }
        
        error_page   500 502 503 504  /50x.html;
        location = /50x.html {
                root   /usr/share/nginx/html;
        }

}
